-- =========================================================
-- CREAR BASE DE DATOS
-- =========================================================
CREATE DATABASE vacacionesgyj;
GO

USE vacacionesgyj;
GO

-- =========================================================
-- TABLA: centros_operacion
-- =========================================================
CREATE TABLE centros_operacion (
    id INT IDENTITY(1,1) PRIMARY KEY,
    nombre VARCHAR(150) NOT NULL,
    estado BIT NOT NULL DEFAULT 1,
    fecha_registro DATETIME NOT NULL DEFAULT GETDATE(),
    fecha_update DATETIME NULL
);

-- =========================================================
-- TABLA: inf_asignacion
-- =========================================================
CREATE TABLE inf_asignacion (
    id INT IDENTITY(1,1) PRIMARY KEY,

    identificacion INT NULL,

    empleado VARCHAR(200) NOT NULL,

    cargo VARCHAR(200) NOT NULL,

    desc_unes VARCHAR(200) NOT NULL,

    unes VARCHAR(200) NOT NULL,

    correo VARCHAR(200) NOT NULL,

    estado_asignacion VARCHAR(200) NOT NULL,

    dias_a_corte FLOAT NOT NULL,

    observaciones VARCHAR(200) NOT NULL,

    motivo_rechazo VARCHAR(500) NULL,

    -- PERIODO 1
    periodo1_fecha1 DATETIME NULL,
    periodo1_fecha2 DATETIME NULL,
    periodo1_dias1 INT NULL,

    -- PERIODO 2
    periodo2_fecha1 DATETIME NULL,
    periodo2_fecha2 DATETIME NULL,
    periodo2_dias2 INT NULL,

    -- PERIODO 3
    periodo3_fecha1 DATETIME NULL,
    periodo3_fecha2 DATETIME NULL,
    periodo3_dias3 INT NULL,

    fecha_regreso DATETIME NOT NULL,

    total_dias_empleado FLOAT NULL,

    fecha_registro DATETIME NOT NULL DEFAULT GETDATE(),

    fecha_update DATETIME NULL
);
GO

-- =========================================================
-- TABLA: notificaciones
-- =========================================================
CREATE TABLE notificaciones (
    id INT IDENTITY(1,1) PRIMARY KEY,

    unes_notificacion INT NULL,

    titulo VARCHAR(200) NOT NULL,

    mensaje VARCHAR(MAX) NOT NULL,

    leida BIT NOT NULL DEFAULT 0,

    fecha_notificacion DATETIME NOT NULL DEFAULT GETDATE()
);
GO

-- =========================================================
-- ÍNDICES
-- =========================================================

CREATE INDEX IX_inf_asignacion_identificacion
ON inf_asignacion(identificacion);
GO

CREATE INDEX IX_inf_asignacion_unes
ON inf_asignacion(unes);
GO

CREATE INDEX IX_notificaciones_unes_notificacion
ON notificaciones(unes_notificacion);
GO
